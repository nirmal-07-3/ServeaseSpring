public interface BookingRepository
        extends JpaRepository<Booking, Long> {

    /*
     * Get all bookings of a customer.
     */
    List<Booking> findByCustomerOrderByCreatedAtDesc(
            User customer
    );

    /*
     * Find a particular booking belonging to a customer.
     */
    Optional<Booking> findByIdAndCustomer(
            Long id,
            User customer
    );

    /*
     * Get bookings for a particular provider service.
     */
    List<Booking> findByProviderServiceOrderByCreatedAtDesc(
            ProviderService providerService
    );

    /*
     * Get all bookings belonging to a provider.
     *
     * Provider is derived through ProviderService.
     */
    List<Booking> findByProviderServiceProviderOrderByCreatedAtDesc(
            ProviderProfile provider
    );

    /*
     * Find a booking belonging to a provider.
     */
    Optional<Booking> findByIdAndProviderServiceProvider(
            Long id,
            ProviderProfile provider
    );

    /*
     * Check whether a provider has an active booking
     * at the requested date and time.
     *
     * We will use this for double-booking prevention.
     */
    List<Booking> findByProviderServiceProviderAndBookingDateAndStatusIn(
            ProviderProfile provider,
            LocalDate bookingDate,
            List<BookingStatus> statuses
    );

    /*
     * Get all bookings with a particular status.

     */

            BookingStatus status

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.DayOfWeek;
import java.time.LocalTime;
import java.util.List;

@Service
@RequiredArgsConstructor
@Transactional
public class BookingServiceImpl implements BookingService {

    private final BookingRepository bookingRepository;

    private final ProviderServiceRepository providerServiceRepository;

    private final ProviderAvailabilityRepository providerAvailabilityRepository;

    private final BookingMapper bookingMapper;

    private final UserService userService;

    @Override
    public BookingResponse createBooking(
            CreateBookingRequest request) {

        // 1. Get authenticated customer
        User customer = userService.getAuthenticatedUser();

        // 2. Find active provider service
        ProviderService providerService =
                providerServiceRepository
                        .findByIdAndActiveTrue(
                                request.getProviderServiceId()
                        )
                        .orElseThrow(() ->
                                new BookingNotFoundException(
                                        "Active provider service not found."
                                ));

        // 3. Get provider from the service
        ProviderProfile provider =
                providerService.getProvider();

        // 4. Provider must be approved
        if (provider.getVerificationStatus()
                != VerificationStatus.APPROVED) {

            throw new BookingConflictException(
                    "This provider is not currently available for booking."
            );
        }

        // 5. Check provider availability
        DayOfWeek dayOfWeek =
                request.getBookingDate().getDayOfWeek();

        ProviderAvailability availability =
                providerAvailabilityRepository
                        .findByProviderAndDayOfWeek(
                                provider,
                                dayOfWeek
                        )
                        .orElseThrow(() ->
                                new BookingConflictException(
                                        "Provider has not configured availability for this day."
                                ));

        // 6. Check holiday
        if (!Boolean.TRUE.equals(
                availability.getAvailable())) {

            throw new BookingConflictException(
                    "Provider is not available on the selected day."
            );
        }

        // 7. Check working hours
        LocalTime bookingTime =
                request.getBookingTime();

        if (availability.getStartTime() == null
                || availability.getEndTime() == null) {

            throw new BookingConflictException(
                    "Provider has not configured working hours for this day."
            );
        }

        if (bookingTime.isBefore(
                availability.getStartTime())
                ||
                bookingTime.isAfter(
                        availability.getEndTime())) {

            throw new BookingConflictException(
                    "Selected time is outside the provider's working hours."
            );
        }

        // 8. Check existing active booking
        // 8. Check for overlapping active bookings

        List<BookingStatus> activeStatuses =
                List.of(
                        BookingStatus.PENDING,
                        BookingStatus.ACCEPTED,
                        BookingStatus.IN_PROGRESS
                );

        List<Booking> existingBookings =
                bookingRepository
                        .findByProviderServiceProviderAndBookingDateAndStatusIn(
                                provider,
                                request.getBookingDate(),
                                activeStatuses
                        );

        LocalTime requestedStart =
                request.getBookingTime();

        LocalTime requestedEnd =
                requestedStart.plusMinutes(
                        providerService
                                .getEstimatedDurationMinutes()
                );

        for (Booking existingBooking : existingBookings) {

            LocalTime existingStart =
                    existingBooking.getBookingTime();

            LocalTime existingEnd =
                    existingStart.plusMinutes(
                            existingBooking
                                    .getProviderService()
                                    .getEstimatedDurationMinutes()
                    );

            boolean overlaps =
                    requestedStart.isBefore(existingEnd)
                            &&
                            requestedEnd.isAfter(existingStart);

            if (overlaps) {

                throw new BookingConflictException(
                        "Provider already has a booking during the selected time."
                );
            }
        }

        // 9. Create booking
        Booking booking = Booking.builder()
                .customer(customer)
                .providerService(providerService)
                .bookingDate(request.getBookingDate())
                .bookingTime(request.getBookingTime())

                // Snapshot current service price
                .agreedPrice(
                        providerService.getBasePrice()
                )

                .address(request.getAddress())
                .city(request.getCity())
                .state(request.getState())
                .pincode(request.getPincode())
                .customerNote(request.getCustomerNote())

                .status(BookingStatus.PENDING)

                .build();

        Booking saved =
                bookingRepository.save(booking);

        return bookingMapper.toResponse(saved);
    }

    @Transactional(readOnly = true)

    }


    @Override
    public void cancelBooking(Long id) {

        User customer = userService.getAuthenticatedUser();

        Booking booking =
                bookingRepository
                        .findByIdAndCustomer(id, customer)
                        .orElseThrow(() ->
                                new BookingNotFoundException(
                                        "Booking not found."
                                ));

        if (booking.getStatus() != BookingStatus.PENDING
                && booking.getStatus() != BookingStatus.ACCEPTED) {

            throw new InvalidBookingStatusException(
                    "This booking cannot be cancelled in its current status."
            );
        }

        booking.setStatus(BookingStatus.CANCELLED);

        bookingRepository.save(booking);
    }

        User user = userService.getAuthenticatedUser();

        ProviderProfile provider = user.getProviderProfile();

        if (provider == null) {
            throw new ProviderNotFoundException(
                    "Provider profile not found."
            );
        }

        return provider;
    }

    @PostMapping("/razorpay/webhook")

            // Raw Razorpay webhook JSON
            @RequestBody(required = false)
            String payload,

            // Razorpay webhook signature
            @RequestHeader(
                    value = "X-Razorpay-Signature",
                    required = false
            )
            String signature,

            // Unique Razorpay webhook event ID
            @RequestHeader(
                    value = "x-razorpay-event-id",
                    required = false
            )
            String eventId
    ) {

        // -------------------------------------------------
        // Check payload
        // -------------------------------------------------

        if (payload == null || payload.isBlank()) {

            return ResponseEntity
                    .badRequest()
                    .body("Missing webhook payload.");
        }


        // -------------------------------------------------
        // Check signature
        // -------------------------------------------------

        if (signature == null || signature.isBlank()) {

            return ResponseEntity
                    .badRequest()
                    .body("Missing Razorpay webhook signature.");
        }


        // -------------------------------------------------
        // Check event ID
        // -------------------------------------------------

        if (eventId == null || eventId.isBlank()) {

            return ResponseEntity
                    .badRequest()
                    .body("Missing Razorpay event ID.");
        }


        // -------------------------------------------------
        // Process webhook
        // -------------------------------------------------

        razorpayWebhookService.processWebhook(
                payload,
                signature,
                eventId
        );


        // -------------------------------------------------
        // Success response
        // -------------------------------------------------

        return ResponseEntity.ok(
                "Webhook received."
        );
    }
}

import org.hibernate.annotations.UpdateTimestamp;

import java.time.LocalDateTime;

@Entity
@Table(
        name = "payments",
        uniqueConstraints = {
                @UniqueConstraint(
                        columnNames = "booking_id"
                )
        }
)
@Getter
@Setter
@NoArgsConstructor
@AllArgsConstructor
@Builder
public class Payment {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    /*
     * One booking should have one payment record.
     */
    @OneToOne(fetch = FetchType.LAZY)

            name = "booking_id",

     * Amount that the customer is expected to pay.

     */
    @Column(
            nullable = false,
            precision = 10,
            scale = 2
    )

    @Builder.Default
    private PaymentStatus status = PaymentStatus.PENDING;

    @Enumerated(EnumType.STRING)
    @Column(nullable = false)
    private PaymentMethod paymentMethod;

    /*
     * ID returned by the payment gateway.
     *
     * For example, Razorpay payment/order ID later.
     */
    @Column(unique = true)
    private String transactionId;

    /*
     * Optional gateway order ID.
     *
     * Useful when integrating Razorpay later.

                bookingRepository
                        .findByIdAndCustomer(

                .amount(booking.getAgreedPrice())

                .paymentMethod(
                        request.getPaymentMethod()
                )

                .status(PaymentStatus.PENDING)

                paymentRepository.save(payment);

        implements RazorpayPaymentService {

    private final BookingRepository bookingRepository;

    private final PaymentRepository paymentRepository;

    private final UserService userService;

    private final PaymentMapper paymentMapper;

    @Value("${razorpay.key.id}")
    private String razorpayKeyId;

    @Value("${razorpay.key.secret}")
    private String razorpayKeySecret;

    @Override
    public RazorpayOrderResponse createOrder(
            Long bookingId) {

        // 1. Get authenticated customer
        User customer =
                userService.getAuthenticatedUser();

        // 2. Find customer's booking
        Booking booking =
                bookingRepository
                        .findByIdAndCustomer(
                                bookingId,
                                customer
                        )
                        .orElseThrow(() ->
                                new BookingNotFoundException(
                                        "Booking not found."
                                ));

        // 3. Booking must be accepted
        if (booking.getStatus()
                != BookingStatus.ACCEPTED) {

            throw new PaymentConflictException(
                    "Payment can only be created for an accepted booking."
            );
        }

        // 4. Check existing payment
        Payment payment =
                paymentRepository
                        .findByBookingId(bookingId)
                        .orElseThrow(() ->
                                new PaymentConflictException(
                                        "Payment record not found. Create the payment first."
                                ));

        // 5. Don't create another Razorpay order
        if (payment.getGatewayOrderId() != null) {

            throw new PaymentConflictException(
                    "Razorpay order already exists for this payment."
            );
        }

        // 6. Get amount from Booking
        BigDecimal amount =
                booking.getAgreedPrice();

        // Razorpay expects amount in paise
        long amountInPaise =
                amount
                        .multiply(BigDecimal.valueOf(100))
                        .longValueExact();

        // 7. Prepare Razorpay order request
        JSONObject orderRequest =
                new JSONObject();

        orderRequest.put(
                "amount",
                amountInPaise
        );

        orderRequest.put(
                "currency",
                "INR"
        );

        orderRequest.put(
                "receipt",
                "SERVEASE_" + booking.getId()
        );

        // 8. Create Razorpay order
        Order razorpayOrder;

        try {

            razorpayOrder =
                    razorpayClient.orders.create(
                            orderRequest
                    );

        } catch (Exception ex) {

            throw new PaymentConflictException(
                    "Unable to create Razorpay order."
            );
        }



        // 9. Save Razorpay order ID
        String razorpayOrderId =
                razorpayOrder.get("id");

        payment.setGatewayOrderId(
                razorpayOrderId
        );

        payment.setStatus(
                PaymentStatus.PENDING
        );

        Payment savedPayment =
                paymentRepository.save(payment);

        // 10. Return order details
        return RazorpayOrderResponse.builder()
                .paymentId(savedPayment.getId())
                .bookingId(booking.getId())
                .razorpayOrderId(
                        razorpayOrderId
                )
                .razorpayKeyId(
                        razorpayKeyId
                )
                .amount(amount)
                .currency("INR")
                .status("created")
                .build();
    }


    @Override
    public PaymentResponse verifyPayment(
            VerifyPaymentRequest request) {

        // 1. Find our payment using Razorpay Order ID
        Payment payment =
                paymentRepository
                        .findByGatewayOrderId(
                                request.getRazorpayOrderId()
                        )
                        .orElseThrow(() ->
                                new PaymentNotFoundException(
                                        "Payment not found for this Razorpay order."
                                ));

        // 2. Get authenticated customer
        User customer =
                userService.getAuthenticatedUser();

        // 3. Make sure this payment belongs to
        //    the authenticated customer's booking
        if (!payment.getBooking()
                .getCustomer()
                .getId()
                .equals(customer.getId())) {

            throw new PaymentConflictException(
                    "You are not authorized to verify this payment."
            );
        }

        // 4. Prevent re-processing an already successful payment
        if (payment.getStatus()
                == PaymentStatus.SUCCESS) {

            return paymentMapper.toResponse(payment);
        }

        /*
         * 5. Verify Razorpay signature.
         *
         * Razorpay signature verification uses:
         *
         * order_id + "|" + payment_id
         *
         * together with your Razorpay Key Secret.
         */
        try {

            String payload =
                    request.getRazorpayOrderId()
                            + "|"
                            + request.getRazorpayPaymentId();

            Utils.verifySignature(
                    payload,
                    request.getRazorpaySignature(),
                    razorpayKeySecret
            );

        } catch (Exception ex) {

            throw new PaymentConflictException(
                    "Razorpay payment signature verification failed."
            );
        }

        // 6. Signature is valid
        payment.setTransactionId(
                request.getRazorpayPaymentId()
        );

        payment.setStatus(
                PaymentStatus.SUCCESS
        );

        // 7. Save updated payment
        Payment savedPayment =
                paymentRepository.save(payment);

        return paymentMapper.toResponse(
                savedPayment
        );
    }
}

import tools.jackson.databind.json.JsonMapper;

import java.time.LocalDateTime;

@Service
@RequiredArgsConstructor
public class RazorpayWebhookServiceImpl
        implements RazorpayWebhookService {





    private final PaymentRepository paymentRepository;

    private final JsonMapper jsonMapper;

    private final RazorpayWebhookEventRepository webhookEventRepository;

    @Value("${razorpay.webhook.secret}")
    private String webhookSecret;


    @Override
    @Transactional
    public void processWebhook(
            String payload,
            String signature,
            String eventId
    ) {

        // 1. Verify Razorpay signature

        try {

            Utils.verifyWebhookSignature(
                    payload,
                    signature,
                    webhookSecret
            );

        } catch (Exception e) {

            throw new IllegalArgumentException(
                    "Invalid Razorpay webhook signature."
            );
        }


        // 2. Check duplicate event

        if (webhookEventRepository.existsByEventId(eventId)) {
            return;
        }


        try {

            JsonNode root =
                    jsonMapper.readTree(payload);

            String event =
                    root.path("event")
                            .asText();


            // 3. Store event ID BEFORE processing

            RazorpayWebhookEvent webhookEvent =
                    RazorpayWebhookEvent.builder()
                            .eventId(eventId)
                            .eventType(event)
                            .receivedAt(LocalDateTime.now())
                            .build();

            System.out.println("Saving webhook event: " + eventId);

            webhookEventRepository.save(webhookEvent);


            // 4. Handle payment.captured

            if ("payment.captured".equals(event)) {

                JsonNode paymentEntity =
                        root.path("payload")
                                .path("payment")
                                .path("entity");

                String razorpayPaymentId =
                        paymentEntity
                                .path("id")
                                .asText();

                String razorpayOrderId =
                        paymentEntity
                                .path("order_id")
                                .asText();


                if (razorpayOrderId.isBlank()) {

                    throw new IllegalArgumentException(
                            "Razorpay order ID missing from webhook."
                    );
                }


                // 5. Find our payment

                Payment payment =
                        paymentRepository
                                .findByGatewayOrderId(
                                        razorpayOrderId
                                )
                                .orElse(null);


                if (payment == null) {
                    return;
                }


                // 6. Idempotency at payment level

                if (payment.getStatus()
                        == PaymentStatus.SUCCESS) {

                    return;
                }


                // 7. Update payment

                payment.setTransactionId(
                        razorpayPaymentId
                );

                payment.setStatus(
                        PaymentStatus.SUCCESS
                );

                paymentRepository.save(payment);
            }

        } catch (Exception e) {

            throw new IllegalArgumentException(
                    "Unable to process Razorpay webhook.",
                    e
            );
        }
    }
}

                            " document already uploaded."
            );
        }

        // Upload file
        String filePath =
                storageService.uploadFile(
                        file,
                        "providers/documents"
                );

        // Create entity
        ProviderDocument document =
                new ProviderDocument();

        document.setProvider(provider);
        document.setDocumentType(documentType);
        document.setDocumentNumber(
                documentNumber
        );
        document.setDocumentUrl(
                filePath
        );
        document.setVerified(false);

        ProviderDocument savedDocument =
                providerDocumentRepository.save(
                        document
                );

        return providerDocumentMapper.toResponse(
                savedDocument
        );
    }


    // ============================================================
    // GET MY DOCUMENTS
    // ============================================================

    @Override
    public List<UploadDocumentResponse> getMyDocuments() {

        ProviderProfile provider =
                getAuthenticatedProvider();

        return providerDocumentRepository
                .findByProvider(provider)
                .stream()
                .map(providerDocumentMapper::toResponse)
                .toList();
    }


    // ============================================================
    // AVAILABILITY
    // ============================================================

    @Override
    @Transactional
    public void saveOrUpdateAvailability(
            List<SaveAvailabilityRequest> requests) {

        ProviderProfile provider =
                getAuthenticatedProvider();

        for (SaveAvailabilityRequest request :
                requests) {

            ProviderAvailability availability =
                    providerAvailabilityRepository
                            .findByProviderAndDayOfWeek(
                                    provider,
                                    request.getDayOfWeek()
                            )
                            .orElse(
                                    new ProviderAvailability()
                            );

            availability.setProvider(provider);

            providerAvailabilityMapper.updateEntity(
                    request,
                    availability
            );

            if (!request.getAvailable()) {

                availability.setStartTime(null);
                availability.setEndTime(null);

            } else {

                if (
                        request.getStartTime() == null ||
                                request.getEndTime() == null
                ) {

                    throw new InvalidAvailabilityException(
                            "Start time and end time are required."
                    );
                }

                if (
                        !request.getStartTime()
                                .isBefore(
                                        request.getEndTime()
                                )
                ) {

                    throw new InvalidAvailabilityException(
                            "End time must be after start time."
                    );
                }
            }

            providerAvailabilityRepository.save(
                    availability
            );
        }
    }


    @Override
    public List<AvailabilityResponse> getAvailability() {

        ProviderProfile provider =
                getAuthenticatedProvider();

        return providerAvailabilityRepository
                .findByProvider(provider)
                .stream()
                .map(providerAvailabilityMapper::toResponse)
                .toList();
    }


    @Override
    public Optional<ProviderAvailability>
    findByProviderAndDayOfWeek(
            ProviderProfile provider,
            DayOfWeek dayOfWeek) {

        return Optional.empty();
    }


    @Override
    public Optional<ProviderAvailability>
    getAvailabilityForDay(
            ProviderProfile provider,
            DayOfWeek dayOfWeek) {

        return providerAvailabilityRepository
                .findByProviderAndDayOfWeek(
                        provider,
                        dayOfWeek
                );
    }

  // MASK DOCUMENT NUMBER
  // ============================================================

  maskDocumentNumber(
    documentNumber: string
  ): string {

    if (!documentNumber) {

      return '-';

    }


    if (documentNumber.length <= 4) {

      return documentNumber;

    }


    const visible =
      documentNumber.slice(-4);


    return `•••• ${visible}`;

  }

  // ============================================================
  // FORMAT DATE
  // ============================================================

  formatDate(
    value: string
  ): string {

    if (!value) {

      return '-';

    }


    const date =
      new Date(value);


    if (
      Number.isNaN(
        date.getTime()
      )
    ) {

      return value;

    }


    return date.toLocaleDateString(
      'en-IN',
      {
        day: '2-digit',
        month: 'short',
        year: 'numeric'
      }
    );

  }


  // ============================================================
  // FORMAT RATING
  // ============================================================

  formatRating(
    rating: number | null
  ): string {

    if (
      rating === null ||
      rating === undefined
    ) {

      return '0.0';

    }


    return rating.toFixed(1);

  }

      const categoryId =
        Number(params.get('categoryId'));

      const subCategoryId =
        Number(params.get('subCategoryId'));

      this.serviceId = serviceId;


      this.loadService();

      today.getFullYear();

    const month =
      String(
        today.getMonth() + 1
      ).padStart(2, '0');

    const day =
      String(
        today.getDate()
      ).padStart(2, '0');


    this.minDate =
      `${year}-${month}-${day}`;

  }

          else {

            this.error =
              response?.message ||
              'Unable to create booking.';

          }


          this.cdr.detectChanges();

        },


        error: (error) => {

          console.error(
            'BOOKING CREATION ERROR:',
            error
          );


          this.error =
            error?.error?.message ||
            'Unable to create booking right now. Please try again.';


          this.cdr.detectChanges();

        }

      });

  }


  validateForm(): boolean {

    if (!this.bookingDate) {

      this.error =
        'Please select a booking date.';

      return false;

    }


    if (
      this.bookingDate < this.minDate
    ) {

      this.error =
        'Booking date cannot be before today.';

      return false;

    }


    if (!this.bookingTime) {

      this.error =
        'Please select a booking time.';

      return false;

    }


    if (!this.address.trim()) {

      this.error =
        'Please enter your address.';

      return false;

    }


    if (!this.city.trim()) {

      this.error =
        'Please enter your city.';

      return false;

    }


    if (!this.state.trim()) {

      this.error =
        'Please enter your state.';

      return false;

    }


    if (!this.pincode.trim()) {

      this.error =
        'Please enter your pincode.';

      return false;

    }


    if (
      !/^\d{6}$/.test(
        this.pincode.trim()
      )
    ) {

      this.error =
        'Please enter a valid 6-digit pincode.';

      return false;

    }


    return true;

  }


  formatPrice(price: number): string {

    return new Intl.NumberFormat(
      'en-IN',
      {
        maximumFractionDigits: 0
      }
    ).format(price || 0);

  }


  formatDuration(minutes: number): string {

    if (!minutes || minutes <= 0) {

      return 'Duration varies';

    }


    const hours =
      Math.floor(minutes / 60);

    const remainingMinutes =
      minutes % 60;


    if (hours === 0) {

      return `${remainingMinutes} min`;

    }


    if (remainingMinutes === 0) {

      return `${hours} hr`;

    }


    return `${hours} hr ${remainingMinutes} min`;

  }

}

import { CommonModule } from '@angular/common';
import {
  ChangeDetectorRef,
  Component,
  OnInit
} from '@angular/core';
import {
  ActivatedRoute,
  Router,
  RouterLink
} from '@angular/router';
import { finalize } from 'rxjs';

import { Api } from '../../../core/services/api';

interface ApiResponse<T> {
  success: boolean;
  message: string;
  data: T;
}

interface Booking {
  id: number;
  customerId: number;
  providerId: number;
  providerBusinessName: string;
  providerServiceId: number;
  serviceTitle: string;
  subCategoryId: number;
  subCategoryName: string;
  bookingDate: string;
  bookingTime: string;
  agreedPrice: number;
  address: string;
  city: string;
  state: string;
  pincode: string;
  customerNote?: string;
  status:
    | 'PENDING'
    | 'ACCEPTED'
    | 'REJECTED'
    | 'IN_PROGRESS'
    | 'COMPLETED'
    | 'CANCELLED';
}

interface Payment {
  id: number;
  bookingId: number;
  amount: number;
  status:
    | 'PENDING'
    | 'SUCCESS'
    | 'FAILED'
    | 'REFUNDED';
  paymentMethod:
    | 'UPI'
    | 'CARD'
    | 'NET_BANKING'
    | 'CASH';
  transactionId?: string;
  gatewayOrderId?: string;
  createdAt?: string;
  updatedAt?: string;
}

interface RazorpayOrder {
  paymentId: number;
  bookingId: number;
  razorpayOrderId: string;
  razorpayKeyId: string;
  amount: number;
  currency: string;
  status: string;
}

interface RazorpayPaymentResponse {
  razorpay_payment_id: string;
  razorpay_order_id: string;
  razorpay_signature: string;
}

declare global {
  interface Window {
    Razorpay: any;
  }
}

@Component({
  selector: 'app-booking-details',
  standalone: true,
  imports: [
    CommonModule,
    RouterLink
  ],
  templateUrl: './booking-details.html',
  styleUrl: './booking-details.css'
})
export class BookingDetails implements OnInit {

  booking: Booking | null = null;

  loading = false;
  error = '';

  payment: Payment | null = null;
  paymentLoading = false;
  paymentProcessing = false;
  paymentError = '';
  paymentSuccess = '';

  constructor(
    private api: Api,
    private route: ActivatedRoute,
    private router: Router,
    private cdr: ChangeDetectorRef
  ) {}

  ngOnInit(): void {

    const id =
      this.route.snapshot.paramMap.get('id');

    if (!id) {
      this.error = 'Invalid booking ID.';
      return;
    }

    this.loadBooking(Number(id));
  }

  loadBooking(id: number): void {

    this.loading = true;
    this.error = '';

    this.api
      .get<ApiResponse<Booking>>(
        `/bookings/${id}`
      )
      .pipe(
        finalize(() => {
          this.loading = false;
          this.cdr.detectChanges();
        })
      )
      .subscribe({

        next: (response) => {

          this.booking =
            response?.data ?? null;

          if (this.booking) {
            this.loadPayment(this.booking.id);
          }

          this.cdr.detectChanges();
        },

        error: (err) => {

          console.error(
            'Failed to load booking:',
            err
          );

          this.booking = null;

          this.error =
            err?.error?.message ||
            'Unable to load booking details.';

          this.cdr.detectChanges();
        }

      });
  }

  loadPayment(bookingId: number): void {

    this.paymentLoading = true;
    this.paymentError = '';

    this.api
      .get<ApiResponse<Payment>>(
        `/payments/booking/${bookingId}`
      )
      .pipe(
        finalize(() => {
          this.paymentLoading = false;
          this.cdr.detectChanges();
        })
      )
      .subscribe({

        next: (response) => {

          this.payment =
            response?.data ?? null;
        },

        error: (err) => {

          if (err?.status === 404) {
            this.payment = null;
            return;
          }

          console.error(
            'Failed to load payment:',
            err
          );

          this.payment = null;

          this.paymentError =
            err?.error?.message ||
            'Unable to load payment information.';
        }

      });
  }

  canPay(): boolean {

    return (
      !!this.booking &&
      this.booking.status === 'ACCEPTED' &&
      this.payment?.status !== 'SUCCESS' &&
      !this.paymentProcessing
    );
  }

  isPaymentSuccessful(): boolean {

    return this.payment?.status === 'SUCCESS';
  }

  payNow(): void {

    if (!this.booking || !this.canPay()) {
      return;
    }

    this.paymentProcessing = true;
    this.paymentError = '';
    this.paymentSuccess = '';

    const bookingId = this.booking.id;

    const createPayment$ = this.payment
      ? this.api.get<ApiResponse<Payment>>(
        `/payments/booking/${bookingId}`
      )
      : this.api.post<ApiResponse<Payment>>(
        '/payments',
        {
          bookingId,
          paymentMethod: 'UPI'
        }
      );

    createPayment$
      .pipe(
        finalize(() => {
          this.cdr.detectChanges();
        })
      )
      .subscribe({

        next: (response) => {

          this.payment =
            response?.data ?? this.payment;

          this.createRazorpayOrder(bookingId);
        },

        error: (err) => {

          console.error(
            'Failed to create payment:',
            err
          );

          this.paymentProcessing = false;

          this.paymentError =
            err?.error?.message ||
            'Unable to start the payment.';

          this.cdr.detectChanges();
        }

      });
  }

  private createRazorpayOrder(
    bookingId: number
  ): void {

    this.api
      .post<ApiResponse<RazorpayOrder>>(
        `/payments/razorpay/order/${bookingId}`,
        {}
      )
      .pipe(
        finalize(() => {
          this.cdr.detectChanges();
        })
      )
      .subscribe({

        next: (response) => {

          const order = response?.data;

          if (
            !order?.razorpayOrderId ||
            !order?.razorpayKeyId
          ) {

            this.paymentProcessing = false;

            this.paymentError =
              'Razorpay order information is incomplete.';

            this.cdr.detectChanges();

            return;
          }

          this.openRazorpayCheckout(order);
        },

        error: (err) => {

          console.error(
            'Failed to create Razorpay order:',
            err
          );

          this.paymentProcessing = false;

          this.paymentError =
            err?.error?.message ||
            'Unable to create the Razorpay order.';

          this.cdr.detectChanges();
        }

      });
  }

  private openRazorpayCheckout(
    order: RazorpayOrder
  ): void {

    if (!window.Razorpay) {

      this.paymentProcessing = false;

      this.paymentError =
        'Razorpay Checkout could not be loaded. Please refresh and try again.';

      this.cdr.detectChanges();

      return;
    }

    const options = {

      key: order.razorpayKeyId,

      amount:
        Math.round(order.amount * 100),

      currency:
        order.currency || 'INR',

      name: 'Servease',

      description:
        this.booking?.serviceTitle ||
        'Service Booking',

      order_id:
      order.razorpayOrderId,

      prefill: {},

      theme: {
        color: '#111827'
      },

      modal: {

        ondismiss: () => {

          this.paymentProcessing = false;

          this.paymentError =
            'Payment window was closed. You can try again.';

          this.cdr.detectChanges();
        }

      },

      handler: (
        response: RazorpayPaymentResponse
      ) => {

        this.verifyRazorpayPayment(
          response
        );
      }

    };

    try {

      const razorpay =
        new window.Razorpay(options);

      razorpay.on(
        'payment.failed',
        (response: any) => {

          console.error(
            'Razorpay payment failed:',
            response
          );

          this.paymentProcessing = false;

          this.paymentError =
            response?.error?.description ||
            'Razorpay payment failed. Please try again.';

          this.cdr.detectChanges();
        }
      );

      razorpay.open();

    } catch (err) {

      console.error(
        'Unable to open Razorpay Checkout:',
        err
      );

      this.paymentProcessing = false;

      this.paymentError =
        'Unable to open Razorpay Checkout.';

      this.cdr.detectChanges();
    }
  }

  private verifyRazorpayPayment(
    response: RazorpayPaymentResponse
  ): void {

    this.api
      .post<ApiResponse<Payment>>(
        '/payments/razorpay/verify',
        {
          razorpayPaymentId:
          response.razorpay_payment_id,

          razorpayOrderId:
          response.razorpay_order_id,

          razorpaySignature:
          response.razorpay_signature
        }
      )
      .pipe(
        finalize(() => {

          this.paymentProcessing = false;

          this.cdr.detectChanges();
        })
      )
      .subscribe({

        next: (verifyResponse) => {

          this.payment =
            verifyResponse?.data ??
            this.payment;

          this.paymentSuccess =
            'Payment completed successfully.';

          this.paymentError = '';

          if (this.booking) {
            this.loadPayment(
              this.booking.id
            );
          }

          this.cdr.detectChanges();
        },

        error: (err) => {

          console.error(
            'Payment verification failed:',
            err
          );

          this.paymentError =
            err?.error?.message ||
            'Payment was completed, but verification failed. Please contact support before paying again.';

          this.cdr.detectChanges();
        }

      });
  }

  getStatusLabel(
    status: Booking['status']
  ): string {

    switch (status) {

      case 'PENDING':
        return 'Pending';

      case 'ACCEPTED':
        return 'Accepted';

      case 'IN_PROGRESS':
        return 'In Progress';

      case 'COMPLETED':
        return 'Completed';

      case 'REJECTED':
        return 'Rejected';

      case 'CANCELLED':
        return 'Cancelled';

      default:
        return status;
    }
  }

  getStatusClass(
    status: Booking['status']
  ): string {

    return status
      .toLowerCase()
      .replace('_', '-');
  }

  getProgressStep(
    status: Booking['status']
  ): number {

    switch (status) {

      case 'PENDING':
        return 1;

      case 'ACCEPTED':
        return 2;

      case 'IN_PROGRESS':

export class MyBookings implements OnInit {

  bookings: Booking[] = [];

  loading = false;
  error = '';
  cancellingId: number | null = null;

  constructor(
    private api: Api,
    private cdr: ChangeDetectorRef
  ) {}

  ngOnInit(): void {
    this.loadBookings();
  }

  loadBookings(): void {

    this.loading = true;
    this.error = '';

    this.api
      .get<ApiResponse<Booking[]>>('/bookings')
      .pipe(
        finalize(() => {
          this.loading = false;
          this.cdr.detectChanges();
        })
      )
      .subscribe({

        next: (response) => {
          this.bookings = response?.data ?? [];
          this.cdr.detectChanges();
        },

        error: (err) => {
          console.error(
            'Failed to load bookings:',
            err
          );

          this.bookings = [];

          this.error =
            err?.error?.message ||
            'Unable to load your bookings. Please try again.';

          this.cdr.detectChanges();
        }

      });
  }

  getStatusLabel(
    status: Booking['status']
  ): string {

    switch (status) {

      case 'PENDING':
        return 'Pending';

      case 'ACCEPTED':
        return 'Accepted';

      case 'IN_PROGRESS':
        return 'In Progress';

      case 'COMPLETED':
        return 'Completed';

      case 'REJECTED':
        return 'Rejected';

      case 'CANCELLED':
        return 'Cancelled';

      default:
        return status;
    }
  }

  getStatusClass(
    status: Booking['status']
  ): string {

    return status
      .toLowerCase()
      .replace('_', '-');
  }

  canCancel(
    booking: Booking
  ): boolean {

    return booking.status === 'PENDING';
  }

  cancelBooking(
    booking: Booking
  ): void {

    if (
      !this.canCancel(booking) ||
      this.cancellingId !== null
    ) {
      return;
    }

    const confirmed = window.confirm(
      'Are you sure you want to cancel this booking?'
    );

    if (!confirmed) {
      return;
    }

    this.cancellingId = booking.id;

    this.api
      .patch<ApiResponse<void>>(
        `/bookings/${booking.id}/cancel`
      )
      .pipe(
        finalize(() => {
          this.cancellingId = null;
          this.cdr.detectChanges();
        })
      )
      .subscribe({

        next: () => {
          this.loadBookings();
        },

        error: (err) => {
          console.error(
            'Failed to cancel booking:',
            err
          );

          alert(
            err?.error?.message ||
            'Unable to cancel the booking. Please try again.'
          );
        }

      });
  }

  formatDate(
    date: string
  ): string {

    if (!date) {
      return '';
    }

    const parsedDate =
      new Date(`${date}T00:00:00`);

    return parsedDate.toLocaleDateString(
      'en-IN',
      {
        day: 'numeric',
        month: 'short',
        year: 'numeric'
      }
    );
  }

  formatTime(
    time: string
  ): string {

    if (!time) {
      return '';
    }

    const [hours, minutes] =
      time.split(':');

    const date = new Date();

    date.setHours(
      Number(hours),
      Number(minutes),
      0,
      0
    );

    return date.toLocaleTimeString(
      'en-IN',
      {
        hour: 'numeric',
        minute: '2-digit',
        hour12: true
      }
    );
  }

  getProgressStep(
    status: Booking['status']
  ): number {

    switch (status) {

      case 'PENDING':
        return 1;

      case 'ACCEPTED':
        return 2;

      case 'IN_PROGRESS':
        return 3;

      case 'COMPLETED':
        return 4;

      default:
        return 0;
    }
  }
}

import {
  ChangeDetectorRef,
  Component,
  OnInit
} from '@angular/core';

import {
  CommonModule
} from '@angular/common';

import {
  ActivatedRoute,
  RouterLink
} from '@angular/router';

import {
  Api
} from '../../../core/services/api';

import {
  finalize
} from 'rxjs/operators';


interface ProviderService {

  id: number;

  providerId: number;

  providerBusinessName: string;

  subCategoryId: number;

  subCategoryName: string;

  title: string;

  description: string;

  basePrice: number;

  estimatedDurationMinutes: number;

  serviceRadiusKm: number;

  active: boolean;

}


interface ApiResponse<T> {

  success: boolean;

  message: string;

  data: T;

}


@Component({

  selector: 'app-provider-services',

  standalone: true,

  imports: [
    CommonModule,
    RouterLink
  ],

  templateUrl: './provider-services.html',

  styleUrl: './provider-services.css'

})


export class ProviderServices implements OnInit {


  categoryId!: number;

  subCategoryId!: number;


  subCategoryName = 'Services';


  services: ProviderService[] = [];


  loading = false;

  error = '';


  constructor(

    private route: ActivatedRoute,

    private api: Api,

    private cdr: ChangeDetectorRef

  ) {}


  ngOnInit(): void {

    this.route.paramMap.subscribe(params => {

      const categoryId =
        Number(params.get('categoryId'));

      const subCategoryId =
        Number(params.get('subCategoryId'));


      if (
        !categoryId ||
        categoryId <= 0 ||
        !subCategoryId ||
        subCategoryId <= 0
      ) {

        this.error =
          'Invalid service selection.';

        return;

      }


      this.categoryId = categoryId;

      this.subCategoryId = subCategoryId;


      this.loadServices();

    });

  }


  loadServices(): void {

    this.loading = true;

    this.error = '';

    this.services = [];


    this.api

      .get<ApiResponse<ProviderService[]>>(
        `/subcategories/${this.subCategoryId}/services`
      )

      .pipe(

        finalize(() => {

          this.loading = false;

          this.cdr.detectChanges();

        })

      )

      .subscribe({

        next: (response) => {

          console.log(
            'PROVIDER SERVICES RESPONSE:',
            response
          );


          if (

            response &&

            response.success &&

            Array.isArray(response.data)

          ) {

            this.services =
              response.data;


            if (
              this.services.length > 0 &&
              this.services[0].subCategoryName
            ) {

              this.subCategoryName =
                this.services[0].subCategoryName;

            }

          }

          else {

            this.error =
              response?.message ||
              'Unable to load available services.';

          }


          this.cdr.detectChanges();

        },


        error: (error) => {

          console.error(
            'PROVIDER SERVICES ERROR:',
            error
          );


          this.services = [];


          this.error =
            error?.error?.message ||
            'Unable to load services right now. Please try again.';


          this.cdr.detectChanges();

        }

      });

  }


  retry(): void {

    if (this.loading) {

      return;

    }

    this.loadServices();

  }


  formatPrice(price: number): string {

    return new Intl.NumberFormat(
      'en-IN',
      {
        maximumFractionDigits: 0
      }
    ).format(price || 0);

  }


  formatDuration(minutes: number): string {

    if (!minutes || minutes <= 0) {

      return 'Duration varies';

    }


    const hours =
      Math.floor(minutes / 60);

    const remainingMinutes =
      minutes % 60;


    if (hours === 0) {

      return `${remainingMinutes} min`;

    }


    if (remainingMinutes === 0) {

      return `${hours} hr`;

    }


    return `${hours} hr ${remainingMinutes} min`;

  }


  formatRadius(radius: number): string {

    if (
      radius === null ||
      radius === undefined
    ) {

      return 'Service area varies';

    }


    return `Up to ${radius} km`;

  }

}

import {
  ChangeDetectorRef,
  Component,
  OnInit
} from '@angular/core';

import { CommonModule } from '@angular/common';
import { RouterLink } from '@angular/router';
import { HttpClient } from '@angular/common/http';
import { finalize } from 'rxjs';

export enum BookingStatus {
  PENDING = 'PENDING',
  ACCEPTED = 'ACCEPTED',
  REJECTED = 'REJECTED',
  IN_PROGRESS = 'IN_PROGRESS',
  COMPLETED = 'COMPLETED',
  CANCELLED = 'CANCELLED'
}

export interface BookingResponse {
  id: number;
  customerId: number;
  providerId: number;
  providerBusinessName: string;
  providerServiceId: number;
  serviceTitle: string;
  subCategoryId: number;
  subCategoryName: string;
  bookingDate: string;
  bookingTime: string;
  agreedPrice: number;
  address: string;
  city: string;
  state: string;
  pincode: string;
  customerNote: string | null;
  status: BookingStatus;
}

interface ApiResponse<T> {
  success: boolean;
  message: string;
  data: T;
}

@Component({
  selector: 'app-provider-bookings',
  standalone: true,
  imports: [
    CommonModule,
    RouterLink
  ],
  templateUrl: './bookings.html',
  styleUrl: './bookings.css'
})
export class Bookings implements OnInit {

  bookings: BookingResponse[] = [];

  loading = false;
  actionLoading = false;

  errorMessage = '';
  successMessage = '';

  selectedBooking: BookingResponse | null = null;

  constructor(
    private http: HttpClient,
    private cdr: ChangeDetectorRef
  ) {}

  ngOnInit(): void {
    console.log('PROVIDER BOOKINGS COMPONENT INITIALIZED');

    this.loadBookings();
  }


  // ============================================================
  // LOAD BOOKINGS
  // ============================================================

  loadBookings(): void {

    console.log('LOAD BOOKINGS STARTED');

    this.loading = true;
    this.errorMessage = '';

    this.cdr.detectChanges();

    this.http
      .get<ApiResponse<BookingResponse[]>>(
        '/api/provider/bookings'
      )
      .pipe(
        finalize(() => {

          console.log(
            'LOAD BOOKINGS FINALIZE'
          );

          this.loading = false;

          this.cdr.detectChanges();

          console.log(
            'LOADING AFTER FINALIZE:',
            this.loading
          );
        })
      )
      .subscribe({

        next: (response) => {

          console.log(
            'PROVIDER BOOKINGS RESPONSE:',
            response
          );

          if (
            response &&
            response.success === true &&
            Array.isArray(response.data)
          ) {

            this.bookings = response.data;

            console.log(
              'BOOKINGS ARRAY:',
              this.bookings
            );

          } else {

            this.bookings = [];

            this.errorMessage =
              response?.message ||
              'Unable to load bookings.';
          }

          /*
           * IMPORTANT:
           * Force Angular to render the new state.
           */
          this.loading = false;

          this.cdr.detectChanges();

          console.log(
            'LOADING AFTER SUCCESS:',
            this.loading
          );
        },

        error: (error) => {

          console.error(
            'PROVIDER BOOKINGS ERROR:',
            error
          );

          this.bookings = [];

          this.errorMessage =
            error?.error?.message ||
            'Failed to load bookings. Please try again.';

          this.loading = false;

          this.cdr.detectChanges();

          console.log(
            'LOADING AFTER ERROR:',
            this.loading
          );
        }

      });
  }


  // ============================================================
  // REFRESH
  // ============================================================

  refreshBookings(): void {

    if (this.loading || this.actionLoading) {
      return;
    }

    this.successMessage = '';

    this.loadBookings();
  }


  // ============================================================
  // ACCEPT
  // ============================================================

  acceptBooking(
    booking: BookingResponse
  ): void {

    this.performAction(
      booking,
      'accept',
      'Booking accepted successfully.'
    );
  }


  // ============================================================
  // REJECT
  // ============================================================

  rejectBooking(
    booking: BookingResponse
  ): void {

    this.performAction(
      booking,
      'reject',
      'Booking rejected successfully.'
    );
  }


  // ============================================================
  // START
  // ============================================================

  startBooking(
    booking: BookingResponse
  ): void {

    this.performAction(
      booking,
      'start',
      'Booking started successfully.'
    );
  }


  // ============================================================
  // COMPLETE
  // ============================================================

  completeBooking(
    booking: BookingResponse
  ): void {

    this.performAction(
      booking,
      'complete',
      'Booking completed successfully.'
    );
  }


  // ============================================================
  // COMMON ACTION
  // ============================================================

  private performAction(
    booking: BookingResponse,
    action: string,
    successMessage: string
  ): void {

    if (this.actionLoading) {
      return;
    }

    this.actionLoading = true;
    this.errorMessage = '';
    this.successMessage = '';

    this.http
      .patch<ApiResponse<void>>(
        `/api/provider/bookings/${booking.id}/${action}`,
        {}
      )
      .pipe(
        finalize(() => {

          this.actionLoading = false;

          this.cdr.detectChanges();
        })
      )
      .subscribe({

        next: (response) => {

          if (response?.success) {

            this.successMessage =
              response.message ||
              successMessage;

            /*
             * Reload bookings after successful action.
             */
            this.loadBookings();

          } else {

            this.errorMessage =
              response?.message ||
              `Unable to ${action} booking.`;
          }

          this.cdr.detectChanges();
        },

        error: (error) => {

          console.error(
            `PROVIDER BOOKING ${action.toUpperCase()} ERROR:`,
            error
          );

          this.errorMessage =
            error?.error?.message ||
            `Failed to ${action} booking. Please try again.`;

          this.cdr.detectChanges();
        }

      });
  }


  // ============================================================
  // VIEW DETAILS
  // ============================================================

  viewBooking(
    booking: BookingResponse
  ): void {

    this.selectedBooking = booking;

    this.cdr.detectChanges();
  }


  closeBooking(): void {

    this.selectedBooking = null;

    this.cdr.detectChanges();
  }


  // ============================================================
  // STATUS
  // ============================================================

  getStatusLabel(
    status: BookingStatus
  ): string {

    switch (status) {

      case BookingStatus.PENDING:
        return 'Pending';

      case BookingStatus.ACCEPTED:
        return 'Accepted';

      case BookingStatus.REJECTED:
        return 'Rejected';

      case BookingStatus.IN_PROGRESS:
        return 'In progress';

      case BookingStatus.COMPLETED:
        return 'Completed';

      case BookingStatus.CANCELLED:
        return 'Cancelled';

      default:
        return status;
    }
  }


  getStatusClass(
    status: BookingStatus
  ): string {

    switch (status) {

      case BookingStatus.PENDING:
        return 'pending';

      case BookingStatus.ACCEPTED:
        return 'accepted';

      case BookingStatus.REJECTED:
        return 'rejected';

      case BookingStatus.IN_PROGRESS:
        return 'in-progress';

      case BookingStatus.COMPLETED:
        return 'completed';

      case BookingStatus.CANCELLED:
        return 'cancelled';

      default:
        return '';
    }
  }


  // ============================================================
  // ACTION VISIBILITY
  // ============================================================

  canAccept(
    booking: BookingResponse
  ): boolean {

    return booking.status === BookingStatus.PENDING;
  }


  canReject(
    booking: BookingResponse
  ): boolean {

    return booking.status === BookingStatus.PENDING;
  }


  canStart(
    booking: BookingResponse
  ): boolean {

    return booking.status === BookingStatus.ACCEPTED;
  }


  canComplete(
    booking: BookingResponse
  ): boolean {

    return booking.status === BookingStatus.IN_PROGRESS;
  }


  // ============================================================
  // DATE
  // ============================================================

  formatDate(
    date: string
  ): string {

    if (!date) {
      return '-';
    }

    const parsed = new Date(
      `${date}T00:00:00`
    );

    return parsed.toLocaleDateString(
      'en-IN',
      {
        day: '2-digit',
        month: 'short',
        year: 'numeric'
      }
    );
  }


  // ============================================================
  // TIME
  // ============================================================

  formatTime(
    time: string
  ): string {

    if (!time) {
      return '-';
    }

    const parts = time.split(':');

    if (parts.length < 2) {
      return time;
    }

    const hours = Number(parts[0]);
    const minutes = Number(parts[1]);

    const date = new Date();

    date.setHours(hours);
    date.setMinutes(minutes);
    date.setSeconds(0);

    return date.toLocaleTimeString(
      'en-IN',
      {
        hour: '2-digit',
        minute: '2-digit'
      }
    );
  }


  // ============================================================
  // PRICE
  // ============================================================

  formatPrice(
    price: number
  ): string {

    return new Intl.NumberFormat(
      'en-IN',
      {
        minimumFractionDigits: 2,
        maximumFractionDigits: 2
      }
    ).format(price || 0);
  }


  // ============================================================
  // STATISTICS
  // ============================================================

  get totalBookings(): number {

    return this.bookings.length;
  }


  get pendingBookings(): number {

    return this.bookings.filter(
      booking =>
        booking.status === BookingStatus.PENDING
    ).length;
  }


  get acceptedBookings(): number {

    return this.bookings.filter(
      booking =>
        booking.status === BookingStatus.ACCEPTED
    ).length;
  }


  get completedBookings(): number {

    return this.bookings.filter(
      booking =>
        booking.status === BookingStatus.COMPLETED
    ).length;
  }

}
